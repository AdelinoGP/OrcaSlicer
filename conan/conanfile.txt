# Consolidated dependency manifest — for lockfile generation and direct
# `conan install` workflows (CI cache warming, offline preparation).
#
# KEEP IN SYNC with the add_requires set in xmake.lua. xmake performs one
# `conan install` per package (each with its own generated conanfile.txt), so
# this consolidated file is not what xmake executes — it exists so a single
# `conan lock create` can pin the whole version/revision set:
#
#   conan lock create conan/conanfile.txt \
#       --profile:host=conan/profile_host.txt \
#       --profile:build=conan/profile_build.txt \
#       --lockfile-out=conan/conan.lock
#
# Windows-first: options mirror the Windows configuration in xmake.lua.
# Custom-recipe packages (opencascade 7.6.0, opencv 4.6.0, openvdb 6.2.1-fork,
# tbb 2021.5, openssl 1.1.1w, libcurl 7.75, freetype 2.12.1, opencsg 1.4.2,
# openexr 2.5.5, nlopt 2.5.0) are added here as their recipes land under
# conan/recipes/.
