# Host profile for Orca(pnp_gui) — the settings every dependency must be built with.
# KEEP IN SYNC with the `conan_base` configs in xmake.lua: xmake generates its own
# per-package profiles and applies these same settings as `-s` / `-c` CLI overrides
# (xmake's conan integration cannot consume external profile files or lockfiles).
# This file is for direct conan workflows: `conan lock create`, CI cache warming,
# recipe development under conan/recipes/.
include(default)

[settings]
# The app is C++17; xmake's generated MSVC profile hardcodes cppstd=14, overridden there.
compiler.cppstd=17
{% if platform.system() == "Windows" %}
# DYNAMIC MSVC runtime (/MD) — parity with the authoritative CMake build, which
# never overrides CMAKE_MSVC_RUNTIME_LIBRARY (SLIC3R_STATIC means static
# *libraries*, not static CRT). /MT was tried and reverted: onetbb's recipe
# rejects it (Invalid binary), and shared TBB/OCCT beside a /MT app is a
# cross-heap hazard.
compiler.runtime=dynamic
{% endif %}

[conf]
# VS 18 2026 generator is unknown to cmake < 4 (pinned by e.g. the wxWidgets recipe);
# Ninja works on every cmake version and platform. See handoff §5.1.
tools.cmake.cmaketoolchain:generator=Ninja

[replace_requires]
# Steer every recipe's pinned transitives to the single project-wide versions
# (deps/ exact matches). Effective because the whole graph is resolved by ONE
# conan install of conan/conanfile.py (see ADR-0001 amendment):
#   cgal/5.6.3 pins boost/1.83.0 + eigen/3.4.0
#   opencascade/7.6.0 pins freetype/2.13.2
boost/*: boost/1.84.0
eigen/*: eigen/5.0.1
freetype/*: freetype/2.12.1
